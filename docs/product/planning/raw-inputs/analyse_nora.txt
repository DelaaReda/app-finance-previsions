1. Price features : build_price_features

Hypothèse : tu as un DataFrame de prix daily avec un DatetimeIndex et une colonne "close" (venant de yfinance, FMP, etc.).

# services/features_price.py

from __future__ import annotations
from dataclasses import dataclass
from typing import Dict, Any, Optional
import pandas as pd
import numpy as np


@dataclass
class PriceFeaturesConfig:
    # horizons en jours de bourse (approx)
    horizons: Dict[str, int] = None

    def __post_init__(self):
        if self.horizons is None:
            self.horizons = {
                "1d": 1,
                "5d": 5,
                "1m": 21,
                "3m": 63,
                "6m": 126,
                "1y": 252,
            }


def _compute_return(series: pd.Series, window: int) -> Optional[float]:
    if len(series) <= window:
        return None
    past = series.iloc[-window - 1]
    last = series.iloc[-1]
    if past == 0 or pd.isna(past) or pd.isna(last):
        return None
    return float((last / past) - 1.0)


def _rolling_vol(series: pd.Series, window: int) -> Optional[float]:
    if len(series) < window:
        return None
    returns = series.pct_change().dropna()
    if len(returns) < window:
        return None
    return float(returns.tail(window).std())


def _max_drawdown(series: pd.Series, window: int) -> Optional[float]:
    if len(series) < window:
        return None
    window_series = series.tail(window)
    roll_max = window_series.cummax()
    dd = (window_series / roll_max) - 1.0
    return float(dd.min())


def _trend_state(ret_short: Optional[float], ret_medium: Optional[float]) -> Optional[str]:
    """
    Très simple heuristique : à raffiner plus tard.
    """
    if ret_short is None or ret_medium is None:
        return None
    if ret_short > 0 and ret_medium > 0:
        return "up"
    if ret_short < 0 and ret_medium < 0:
        return "down"
    return "range"


def build_price_features(
    prices: pd.DataFrame,
    config: Optional[PriceFeaturesConfig] = None,
) -> Dict[str, Any]:
    """
    prices: DataFrame avec au minimum colonne 'close' et index datetime trié ascendant.
    """
    if config is None:
        config = PriceFeaturesConfig()

    if "close" not in prices.columns or prices.empty:
        return {
            "price_stats": {},
            "price_profile": {
                "short_term": "Données de prix insuffisantes.",
                "medium_term": "",
                "long_term": "",
            },
        }

    close = prices["close"].astype(float).dropna()
    if close.empty:
        return {
            "price_stats": {},
            "price_profile": {
                "short_term": "Données de prix insuffisantes.",
                "medium_term": "",
                "long_term": "",
            },
        }

    last = float(close.iloc[-1])
    stats: Dict[str, Any] = {"last": last}

    # Returns multi-horizon
    for name, win in config.horizons.items():
        stats[f"ret_{name}"] = _compute_return(close, win)

    # Volatilité
    stats["vol_1m"] = _rolling_vol(close, 21)
    stats["vol_3m"] = _rolling_vol(close, 63)
    stats["vol_1y"] = _rolling_vol(close, 252)

    # Drawdown
    stats["max_drawdown_3m"] = _max_drawdown(close, 63)
    stats["max_drawdown_1y"] = _max_drawdown(close, 252)

    # 1y high/low
    if len(close) >= 252:
        last_1y = close.tail(252)
    else:
        last_1y = close

    high_1y = float(last_1y.max())
    low_1y = float(last_1y.min())
    stats["price_1y_high"] = high_1y
    stats["price_1y_low"] = low_1y
    stats["price_vs_1y_high"] = float(last / high_1y) if high_1y > 0 else None
    stats["price_vs_1y_low"] = float((last - low_1y) / low_1y) if low_1y > 0 else None

    # Momentum simplifié (z-score de ret 1m / 3m / 6m)
    def _z(x):
        if x is None:
            return None
        # bornes pour éviter les extrêmes
        return float(max(min(x, 5.0), -5.0))

    stats["mom_1m"] = _z(stats.get("ret_1m", None))
    stats["mom_3m"] = _z(stats.get("ret_3m", None))
    stats["mom_6m"] = _z(stats.get("ret_6m", None))

    # Trend state
    stats["trend_state_3m"] = _trend_state(stats.get("ret_5d"), stats.get("ret_3m"))

    # Texte profil (condensé pour le LLM)
    def _fmt_pct(v: Optional[float]) -> str:
        if v is None:
            return "N/A"
        return f"{v*100:.1f}%"

    short_term = (
        f"Court terme: { _fmt_pct(stats.get('ret_5d')) } sur 5 jours, "
        f"{ _fmt_pct(stats.get('ret_1m')) } sur 1 mois, "
        f"tendance 3m: { stats.get('trend_state_3m') or 'indéterminée' }."
    )

    medium_term = (
        f"Moyen terme: { _fmt_pct(stats.get('ret_3m')) } sur 3 mois, "
        f"{ _fmt_pct(stats.get('ret_6m')) } sur 6 mois, "
        f"vol 3m≈{ _fmt_pct(stats.get('vol_3m')) }."
    )

    long_term = (
        f"Long terme: { _fmt_pct(stats.get('ret_1y')) } sur 1 an, "
        f"cours actuel ≈ { _fmt_pct(stats.get('price_vs_1y_high')) } du plus haut 1 an."
    )

    price_profile = {
        "short_term": short_term,
        "medium_term": medium_term,
        "long_term": long_term,
    }

    return {
        "price_stats": stats,
        "price_profile": price_profile,
    }

2. Sentiment multi-fenêtres : build_sentiment_features

Hypothèse : tu as déjà news dans ton payload, avec score, sentiment ("positive"/"negative"/"neutral"), et ts/published_at.

# services/features_sentiment.py

from __future__ import annotations
from typing import List, Dict, Any, Optional
from datetime import datetime, timezone, timedelta
import dateutil.parser


def _parse_ts(ts: str) -> Optional[datetime]:
    try:
        dt = dateutil.parser.parse(ts)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        return dt
    except Exception:
        return None


WINDOWS = {
    "24h": timedelta(hours=24),
    "3d": timedelta(days=3),
    "7d": timedelta(days=7),
    "30d": timedelta(days=30),
}


def build_sentiment_features(
    news: List[Dict[str, Any]],
    now: Optional[datetime] = None,
) -> Dict[str, Any]:
    if now is None:
        now = datetime.now(timezone.utc)

    # Normaliser les items news => (dt, score, sentiment)
    items = []
    for n in news or []:
        ts = n.get("ts") or n.get("published_at")
        dt = _parse_ts(ts) if ts else None
        if not dt:
            continue
        score = float(n.get("score", 0.0))
        sentiment = n.get("sentiment")
        items.append((dt, score, sentiment))

    sentiment_windows: Dict[str, Any] = {}

    for label, delta in WINDOWS.items():
        cutoff = now - delta
        subset = [(dt, score, sent) for (dt, score, sent) in items if dt >= cutoff]
        if not subset:
            sentiment_windows[label] = {
                "avg": None,
                "count": 0,
                "pos": 0,
                "neg": 0,
                "neu": 0,
            }
            continue

        scores = [s for _, s, _ in subset]
        avg = sum(scores) / len(scores)

        pos = sum(1 for _, _, s in subset if s == "positive")
        neg = sum(1 for _, _, s in subset if s == "negative")
        neu = sum(1 for _, _, s in subset if s == "neutral")

        sentiment_windows[label] = {
            "avg": avg,
            "count": len(subset),
            "pos": pos,
            "neg": neg,
            "neu": neu,
        }

    # Profil synthétique
    w7 = sentiment_windows.get("7d", {})
    w30 = sentiment_windows.get("30d", {})

    def _trend_7_vs_30():
        a7 = w7.get("avg")
        a30 = w30.get("avg")
        if a7 is None or a30 is None:
            return None
        diff = a7 - a30
        if diff > 5:
            return "improving"
        if diff < -5:
            return "worsening"
        return "stable"

    def _skew_7d():
        if not w7.get("count"):
            return None
        pos = w7.get("pos", 0)
        neg = w7.get("neg", 0)
        if pos >= 2 * max(1, neg):
            return "mostly_positive"
        if neg >= 2 * max(1, pos):
            return "mostly_negative"
        return "mixed"

    sentiment_profile = {
        "trend_7d_vs_30d": _trend_7_vs_30(),
        "skew_7d": _skew_7d(),
        "summary": (
            f"Flux de news: {w7.get('count', 0)} articles/7j, "
            f"sentiment 7j moy≈{w7.get('avg')}, 30j≈{w30.get('avg')}."
        ),
    }

    return {
        "sentiment_windows": sentiment_windows,
        "sentiment_profile": sentiment_profile,
    }

3. Fundamentals + macro : profil rapide

L’idée ici c’est moins des maths, plus de bucketisation pour donner au LLM un contexte lisible.

# services/features_fund_macro.py

from __future__ import annotations
from typing import Dict, Any, Optional


def _bucket_pe(pe: Optional[float]) -> Optional[str]:
    if pe is None:
        return None
    if pe < 10:
        return "very_cheap"
    if pe < 18:
        return "cheap"
    if pe < 25:
        return "fair"
    if pe < 40:
        return "expensive"
    return "very_expensive"


def _bucket_beta(beta: Optional[float]) -> Optional[str]:
    if beta is None:
        return None
    if beta < 0.8:
        return "low_risk"
    if beta < 1.2:
        return "normal_risk"
    if beta < 1.8:
        return "elevated_risk"
    return "high_risk"


def build_fundamentals_profile(fund: Dict[str, Any]) -> Dict[str, Any]:
    pe = fund.get("pe") or fund.get("pe_ratio")
    growth = fund.get("revenueGrowth") or fund.get("revenue_growth")
    margin = fund.get("profitMargins") or fund.get("profit_margin")
    beta = fund.get("beta")

    valuation_bucket = _bucket_pe(pe)
    beta_bucket = _bucket_beta(beta)

    if growth is not None:
        growth_label = "high_growth" if growth > 0.15 else "moderate_growth" if growth > 0.05 else "low_growth"
    else:
        growth_label = None

    if margin is not None:
        margin_label = "high_margin" if margin > 0.3 else "moderate_margin" if margin > 0.1 else "low_margin"
    else:
        margin_label = None

    risk_profile = None
    if beta_bucket == "high_risk" and growth_label == "high_growth":
        risk_profile = "high_beta_high_growth"
    elif beta_bucket in ("elevated_risk", "high_risk"):
        risk_profile = "high_beta"
    elif beta_bucket == "low_risk":
        risk_profile = "defensive"
    else:
        risk_profile = "balanced"

    summary_parts = []
    if valuation_bucket:
        summary_parts.append(f"Valorisation {valuation_bucket.replace('_', ' ')} (PE≈{pe:.1f})." if pe is not None else f"Valorisation {valuation_bucket}.")
    if growth_label and growth is not None:
        summary_parts.append(f"Croissance {growth_label.replace('_', ' ')} (rev_growth≈{growth:.2f}).")
    if margin_label and margin is not None:
        summary_parts.append(f"Margins {margin_label.replace('_', ' ')} (≈{margin:.2f}).")
    if beta_bucket and beta is not None:
        summary_parts.append(f"Risque {beta_bucket.replace('_', ' ')} (beta≈{beta:.2f}).")

    return {
        "valuation": valuation_bucket,
        "growth_label": growth_label,
        "margin_label": margin_label,
        "beta_label": beta_bucket,
        "risk_profile": risk_profile,
        "summary": " ".join(summary_parts),
    }


def build_macro_profile(macro: Dict[str, Any]) -> Dict[str, Any]:
    vix = macro.get("vix")
    us10y = macro.get("us10y")
    dxy = macro.get("dxy")
    vix_delta = macro.get("vix_delta_1m")
    us10y_delta = macro.get("us10y_delta_1m")

    if vix is None:
        vix_level = None
    elif vix < 15:
        vix_level = "low"
    elif vix < 25:
        vix_level = "normal"
    else:
        vix_level = "high"

    if us10y_delta is None:
        us10y_trend = None
    elif us10y_delta > 0.005:
        us10y_trend = "rising"
    elif us10y_delta < -0.005:
        us10y_trend = "falling"
    else:
        us10y_trend = "stable"

    summary_parts = []
    if vix_level:
        summary_parts.append(f"VIX {vix_level} (≈{vix}).")
    if us10y is not None and us10y_trend:
        summary_parts.append(f"Taux US 10y ≈{us10y:.2f}% en tendance {us10y_trend}.")
    if dxy is not None:
        summary_parts.append(f"Dollar US (DXY≈{dxy:.1f}).")

    return {
        "vix_level": vix_level,
        "vix_value": vix,
        "us10y_trend": us10y_trend,
        "dxy_value": dxy,
        "summary": " ".join(summary_parts),
    }

👉 Intégration :

fund_profile = build_fundamentals_profile(features.get("fundamentals", {}) or features.get("fundamentals_enriched", {}))
macro_profile = build_macro_profile(features.get("macro", {}))

features["fundamentals_profile"] = fund_profile
features["macro_profile"] = macro_profile

4. Comment tout brancher dans ton pipeline LLM

Dans ton code qui construit debug_payload["features"] avant l’appel LLM, tu peux faire un truc du style :

features = {
    "ticker": row["ticker"],
    "direction": direction,
    "expected_return": row["expected_return_raw"],
    "expected_return_ensemble": row["expected_return_ensemble"],
    "confidence": row["confidence"],
    "horizon": profile_horizon,
    "tech": tech_block,
    "macro": macro_block,
    "fundamentals": fundamentals_block,
    # etc. (ce que tu as déjà)
}

# 1) Prix (tu fournis df_prices = historique pour ce ticker)
price_feats = build_price_features(df_prices_for_ticker)
features.update(price_feats)

# 2) Sentiment (news déjà filtrées sur ce ticker)
sent_feats = build_sentiment_features(news_for_ticker)
features.update(sent_feats)

# 3) Fundamentals + macro
features["fundamentals_profile"] = build_fundamentals_profile(fundamentals_block)
features["macro_profile"] = build_macro_profile(macro_block)

debug_payload["features"] = features

Ensuite, dans le prompt, tu dis clairement au modèle d’utiliser :
	•	price_profile + price_stats pour le momentum CT/MT/LT,
	•	sentiment_windows + sentiment_profile pour le ton des news,
	•	fundamentals_profile pour le cadre “cher/pas cher, growth/risk”,
	•	macro_profile pour ajuster la prudence.
