2️⃣ Backend Python pour parler à une session tmux (QwenTmuxLLM)

Fichier qwen_tmux_backend.py :

import subprocess
import time
from typing import List, Dict, Optional


class QwenTmuxSession:
    """
    Représente UNE session tmux qui tourne déjà avec `qwen`.
    On envoie du texte avec `tmux send-keys` et on lit la sortie avec `capture-pane`.
    """

    def __init__(self, session_name: str, wait_seconds: float = 2.0):
        self.session = session_name
        self.wait_seconds = wait_seconds
        # snapshot initial pour calculer seulement le "delta" ensuite
        self.last_snapshot = self._capture_full()

    # --- Fonctions internes bas niveau ---

    def _capture_full(self) -> str:
        """Récupère tout le contenu du pane."""
        out = subprocess.check_output(
            ["tmux", "capture-pane", "-pt", self.session],
            text=True,
        )
        return out

    def _get_new_output(self) -> str:
        """Retourne seulement ce qui est nouveau depuis le dernier snapshot."""
        current = self._capture_full()
        # Cas normal: le buffer actuel commence par l'ancien
        if current.startswith(self.last_snapshot):
            new = current[len(self.last_snapshot):]
        else:
            # Si désync (scroll, clear, etc.), on renvoie tout
            new = current
        self.last_snapshot = current
        return new.strip()

    def _send_raw(self, text: str) -> None:
        """Envoie du texte + Enter à la session tmux."""
        cmd = ["tmux", "send-keys", "-t", self.session, text, "C-m"]
        subprocess.check_call(cmd)

    # --- API principale ---

    def ask(self, text: str, wait_seconds: Optional[float] = None) -> str:
        """
        Envoie un message à Qwen dans cette session,
        attend un peu, puis lit la nouvelle sortie.
        """
        if wait_seconds is None:
            wait_seconds = self.wait_seconds

        # envoyer la question
        self._send_raw(text)

        # attendre que Qwen réponde (simple, tu pourras affiner)
        time.sleep(wait_seconds)

        # lire la nouvelle sortie
        answer = self._get_new_output()
        return answer


class QwenTmuxLLM:
    """
    Petit wrapper "LLM-like" au-dessus de QwenTmuxSession.

    - option system_prompt envoyé une fois au début pour "rôler" l'agent.
    - méthode chat(messages) à la sauce OpenAI-style:
      messages = [{"role": "system"|"user"|"assistant", "content": "..."}]
    """

    def __init__(
        self,
        session_name: str,
        system_prompt: str = "",
        wait_seconds: float = 2.0,
    ):
        self.session = QwenTmuxSession(session_name, wait_seconds=wait_seconds)
        self.system_prompt = system_prompt
        self._initialized = False

    def _ensure_initialized(self):
        if self._initialized:
            return
        if self.system_prompt:
            # On envoie le system prompt comme premier message
            self.session.ask(
                f"[SYSTEM ROLE] {self.system_prompt}\n"
                f"Réponds toujours de façon claire. Ne répète pas ce message."
            )
        self._initialized = True

    def chat(self, messages: List[Dict[str, str]]) -> str:
        """
        On envoie à Qwen le DERNIER message user.
        (Simple, mais suffisant comme base.)
        """
        self._ensure_initialized()

        # on récupère le dernier message 'user'
        user_messages = [m for m in messages if m["role"] == "user"]
        if not user_messages:
            raise ValueError("Pas de message user dans messages.")

        last_user = user_messages[-1]["content"]

        # tu pourrais ici reconstruire un historique textuel
        # pour aider Qwen à garder du contexte plus structuré
        prompt = last_user

        answer = self.session.ask(prompt)
        return answer

from qwen_tmux_backend import QwenTmuxLLM


# --- Création des "agents" basés sur Qwen dans tmux ---

planner_llm = QwenTmuxLLM(
    session_name="qwen_planner",
    system_prompt="""Tu es PLANNER.
Tu découpes les demandes de features en tâches techniques claires et numérotées.
Toujours répondre en français, avec une liste d'étapes très concrètes.""",
    wait_seconds=3.0,
)

    session_name="qwen_dev",
    system_prompt="""Tu es DEV.
Tu écris du code et expliques les changements fichier par fichier.
Tu proposes des extraits de code, jamais des gros pavés inutiles.
Toujours répondre en français.""",
    wait_seconds=4.0,
)

tester_llm = QwenTmuxLLM(
    session_name="qwen_tester",
    system_prompt="""Tu es TESTER.
Tu penses comme un QA : tu imagines des tests, détectes les risques,
et donnes un feedback précis au DEV.
Toujours répondre en français.""",
    wait_seconds=3.0,
)


def ask_planner(feature: str) -> str:
    messages = [
        {"role": "user", "content": f"Feature à implémenter : {feature}\nDonne un plan en étapes numérotées."}
    ]
    return planner_llm.chat(messages)


def ask_dev(plan: str) -> str:
    messages = [
        {"role": "user", "content": f"Voici le plan du planner :\n{plan}\nCommence à implémenter la première étape."}
    ]
    return dev_llm.chat(messages)


def ask_tester(dev_output: str) -> str:
    messages = [
        {"role": "user", "content": f"Le développeur a fait ceci :\n{dev_output}\n"
                                   f"Propose des tests et signale les risques potentiels."}
    ]
    return tester_llm.chat(messages)


def main():
    feature = "Ajouter un endpoint /health sur une API FastAPI qui retourne le statut 'ok' et la version."
    print("=== Demande utilisateur ===")
    print(feature)
    print()

    print("=== Planner (Qwen) ===")
    plan = ask_planner(feature)
    print(plan)
    print()

    print("=== Dev (Qwen) ===")
    dev_output = ask_dev(plan)
    print(dev_output)
    print()

    print("=== Tester (Qwen) ===")
    test_feedback = ask_tester(dev_output)
    print(test_feedback)
    print()

    print("=== Fin du mini workflow ===")


if __name__ == "__main__":
    main()

from typing import Any, Dict, List

from autogen import ConversableAgent
from qwen_tmux_backend import QwenTmuxLLM


class QwenTmuxAgent(ConversableAgent):
    """
    Agent AutoGen dont le "cerveau" est QwenTmuxLLM
    qui parle à une session tmux où tourne `qwen`.
    """

    def __init__(
        self,
        name: str,
        session_name: str,
        system_message: str,
        wait_seconds: float = 3.0,
        **kwargs: Any,
    ):
        # On passe un llm_config minimal, mais on ne l'utilisera pas réellement
        super().__init__(
            name=name,
            system_message=system_message,
            llm_config={"config_list": []},  # on bypassera le LLM interne
            **kwargs,
        )
        self.qwen_llm = QwenTmuxLLM(
            session_name=session_name,
            system_prompt=system_message,
            wait_seconds=wait_seconds,
        )

    def generate_reply(
        self,
        messages: List[Dict[str, Any]],
        sender: "ConversableAgent",
        **kwargs: Any,
    ) -> Dict[str, Any]:
        """
        Override de generate_reply pour appeler notre QwenTmuxLLM
        au lieu du LLM par défaut d'AutoGen.
        """
        # On laisse AutoGen construire l'historique complet,
        # mais on ne transmettra à Qwen que le dernier message user envoyé à CET agent.
        # messages = historique complet pour cet agent.
        text = self.qwen_llm.chat(messages)
        return {
            "role": "assistant",
            "content": text,
        }


def build_agents():
    """
    Crée 3 agents AutoGen basés sur 3 sessions tmux différentes :
    - qwen_planner
    - qwen_dev
    - qwen_tester
    """
    planner = QwenTmuxAgent(
        name="planner",
        session_name="qwen_planner",
        system_message=(
            "Tu es PLANNER.\n"
            "Tu découpes les demandes de features en tâches techniques claires, "
            "numérotées, avec un niveau de détail fin. Réponds toujours en français."
        ),
    )

    dev = QwenTmuxAgent(
        name="dev",
        session_name="qwen_dev",
        system_message=(
            "Tu es DEV.\n"
            "Tu écris du code et expliques les changements fichier par fichier. "
            "Tu réponds en français, tu proposes des extraits de code précis."
        ),
    )

    tester = QwenTmuxAgent(
        name="tester",
        session_name="qwen_tester",
        system_message=(
            "Tu es TESTER.\n"
            "Tu raisonnes comme un QA, tu proposes des tests (unitaires, intégration) "
            "et tu signales les risques. Réponds en français."
        ),
    )

    return planner, dev, tester


def simple_workflow():
    """
    Mini workflow AutoGen-like :
    - l'utilisateur décrit une feature
    - planner produit un plan
    - dev propose un début d'implémentation
    - tester propose des tests / risques
    """
    feature = (
        "Ajouter un endpoint /health sur une API FastAPI qui retourne "
        "un JSON avec status='ok' et version='1.0.0'."
    )
    print("=== Feature demandée ===")
    print(feature)
    print()

    planner, dev, tester = build_agents()

    # 1) L'utilisateur parle au planner
    planner_msg = planner.generate_reply(
        messages=[{"role": "user", "content": f"Voici la feature : {feature}\nPropose un plan détaillé."}],
        sender=planner,
    )
    print("=== Réponse du PLANNER (Qwen) ===")
    print(planner_msg["content"])
    print()

    # 2) On donne ce plan au DEV
    dev_msg = dev.generate_reply(
        messages=[
            {
                "role": "user",
                "content": (
                    "Voici le plan donné par le planner :\n\n"
                    f"{planner_msg['content']}\n\n"
                    "Commence à implémenter la première étape et explique les fichiers impactés."
                ),
            }
        ],
        sender=dev,
    )
    print("=== Réponse du DEV (Qwen) ===")
    print(dev_msg["content"])
    print()

    # 3) On donne la sortie du DEV au TESTER
    tester_msg = tester.generate_reply(
        messages=[
            {
                "role": "user",
                "content": (
                    "Le développeur a produit ceci :\n\n"
                    f"{dev_msg['content']}\n\n"
                    "Propose des tests concrets (pytest) et signale les risques éventuels."
                ),
            }
        ],
        sender=tester,
    )
    print("=== Réponse du TESTER (Qwen) ===")
    print(tester_msg["content"])
    print()

    print("=== Fin du workflow simple ===")


if __name__ == "__main__":
    simple_workflow()

        current = self._capture_full()
        # Cas normal: le buffer actuel commence par l'ancien
        if current.startswith(self.last_snapshot):
            new = current[len(self.last_snapshot):]
        else:
            # Si désync (scroll, clear, etc.), on renvoie tout
            new = current
        self.last_snapshot = current
        return new.strip()

    def _send_raw(self, text: str) -> None:
        """Envoie du texte + Enter à la session tmux."""
        cmd = ["tmux", "send-keys", "-t", self.session, text, "C-m"]
        subprocess.check_call(cmd)

    # --- API principale ---

    def ask(self, text: str, wait_seconds: Optional[float] = None) -> str:
        """
        Envoie un message à Qwen dans cette session,
        attend un peu, puis lit la nouvelle sortie.
        """
        if wait_seconds is None:
            wait_seconds = self.wait_seconds

        # envoyer la question
        self._send_raw(text)

        # attendre que Qwen réponde (simple, tu pourras affiner)
        time.sleep(wait_seconds)

        # lire la nouvelle sortie
        answer = self._get_new_output()
        return answer


class QwenTmuxLLM:
    """
    Petit wrapper "LLM-like" au-dessus de QwenTmuxSession.

    messages = [{"role": "system"|"user"|"assistant", "content": "..."}]
    -> on envoie le dernier message user au CLI.
    """

    def __init__(
        self,
        session_name: str,
        system_prompt: str = "",
        wait_seconds: float = 3.0,
    ):
        self.session = QwenTmuxSession(session_name, wait_seconds=wait_seconds)
        self.system_prompt = system_prompt
        self._initialized = False

    def _ensure_initialized(self):
        if self._initialized:
            return
        if self.system_prompt:
            # On envoie le system prompt comme premier message
            self.session.ask(
                f"[SYSTEM ROLE] {self.system_prompt}\n"
                f"Réponds toujours de façon claire. Ne répète pas ce message."
            )
        self._initialized = True

    def chat(self, messages: List[Dict[str, str]]) -> str:
        """
        On envoie à Qwen le DERNIER message user.
        (Simple mais suffisant comme base.)
        """
        self._ensure_initialized()

        user_messages = [m for m in messages if m["role"] == "user"]
        if not user_messages:
            raise ValueError("Pas de message user dans messages.")

        last_user = user_messages[-1]["content"]
        prompt = last_user

        answer = self.session.ask(prompt)
        return answer

import subprocess
import time
from typing import List, Dict, Optional


class QwenTmuxSession:
    def __init__(self, session_name: str, wait_seconds: float = 2.0):
        self.session = session_name
        self.wait_seconds = wait_seconds
        self.last_snapshot = self._capture_full()

    def _capture_full(self) -> str:
        out = subprocess.check_output(
            ["tmux", "capture-pane", "-pt", self.session],
            text=True,
        )
        return out

    def _get_new_output(self) -> str:
        current = self._capture_full()
        if current.startswith(self.last_snapshot):
            new = current[len(self.last_snapshot):]
        else:
            new = current
        self.last_snapshot = current
        return new.strip()

    def _send_raw(self, text: str) -> None:
        subprocess.check_call(
            ["tmux", "send-keys", "-t", self.session, text, "C-m"]
        )

    def ask(self, text: str, wait_seconds: Optional[float] = None) -> str:
        if wait_seconds is None:
            wait_seconds = self.wait_seconds

        self._send_raw(text)
        time.sleep(wait_seconds)
        return self._get_new_output()

class ShellToolAgent(ConversableAgent):
    """
    Agent outil qui exécute VRAIMENT des commandes shell
    dans le répertoire du projet, et renvoie stdout/stderr.
    """

    def __init__(self, name: str = "shell", workdir: str = PROJECT_DIR, **kwargs: Any):
        super().__init__(
            name=name,
            system_message="Tu es un AGENT OUTIL. Tu exécutes des commandes shell reçues et renvoies le résultat.",
            llm_config={"config_list": []},
            human_input_mode="NEVER",
            **kwargs,
        )
        self.workdir = workdir

    def generate_reply(
        self,
        messages: List[Dict[str, Any]],
        sender: "ConversableAgent",
        **kwargs: Any,
    ) -> Dict[str, Any]:
        # On considère que le dernier message contient STRICTEMENT la commande à exécuter
        cmd = messages[-1]["content"].strip()

        try:
            result = subprocess.run(
                cmd,
                shell=True,
                cwd=self.workdir,
                capture_output=True,
                text=True,
                timeout=120,
            )
            output = []
            output.append(f"$ {cmd}")
            output.append("\n--- stdout ---\n")
            output.append(result.stdout or "(vide)")
            output.append("\n--- stderr ---\n")
            output.append(result.stderr or "(vide)")
            output.append(f"\n--- returncode: {result.returncode} ---\n")
            content = "".join(output)
        except Exception as e:

class ShellToolAgent(ConversableAgent):
    """
    Agent outil qui exécute des commandes shell dans PROJECT_DIR.
    La commande est extraite d'une ligne de la forme :
    SHELL_CMD: <commande>
    dans le dernier message.
    """

    def __init__(self, name: str = "shell", workdir: str = PROJECT_DIR, **kwargs: Any):
        super().__init__(
            name=name,
            system_message=(
                "Tu es un AGENT OUTIL shell. "
                "Tu n'es PAS un LLM, tu exécutes juste des commandes reçues."
            ),
            llm_config={"config_list": []},
            human_input_mode="NEVER",
            **kwargs,
        )
        self.workdir = workdir

    def _extract_shell_cmd(self, text: str) -> str | None:
        for line in text.splitlines():
            line = line.strip()
            if line.startswith("SHELL_CMD:"):
                return line[len("SHELL_CMD:"):].strip()
        return None

    def generate_reply(
        self,
        messages: List[Dict[str, Any]],
        sender: "ConversableAgent",
        **kwargs: Any,
    ) -> Dict[str, Any]:
        last = messages[-1]["content"]
        cmd = self._extract_shell_cmd(last)
        if not cmd:
            return {
                "role": "assistant",
                "content": "Aucune ligne 'SHELL_CMD: ...' trouvée dans le message.",
            }

        try:
            result = subprocess.run(
                cmd,
                shell=True,
                cwd=self.workdir,
                capture_output=True,
                text=True,
                timeout=300,
            )
            output = []
            output.append(f"$ {cmd}")
            output.append("\n--- stdout ---\n")
            output.append(result.stdout or "(vide)")
            output.append("\n--- stderr ---\n")
            output.append(result.stderr or "(vide)")
            output.append(f"\n--- returncode: {result.returncode} ---\n")
            content = "".join(output)
        except Exception as e:
            content = f"Erreur lors de l'exécution de la commande `{cmd}` : {e}"

        return {"role": "assistant", "content": content}

class FileToolAgent(ConversableAgent):
    """
    Agent outil pour lire/écrire des fichiers dans PROJECT_DIR.

    Il comprend deux formats dans le dernier message :

    1) Pour écrire un ou plusieurs fichiers :
       FILE:
       path: chemin/relatif.py
       ```python
       ...contenu complet du fichier...
       ```

    2) Pour lire un fichier :
       READ_FILE:
       path: chemin/relatif.py

    Il renvoie un résumé des opérations (écritures + contenus lus).
    """

    FILE_PATTERN = re.compile(
        r"FILE:\s*path:\s*(.+?)\s*```[a-zA-Z0-9_]*\n(.*?)```",
        re.DOTALL,
    )

    READ_PATTERN = re.compile(
        r"READ_FILE:\s*path:\s*(.+)",
        re.IGNORECASE,
    )

    def __init__(self, name: str = "filetool", root_dir: str = PROJECT_DIR, **kwargs: Any):
        super().__init__(
            name=name,
            system_message=(
                "Tu es un AGENT OUTIL fichiers. Tu n'es pas un LLM, tu exécutes des opérations "
                "de lecture/écriture basées sur les instructions reçues."
            ),
            llm_config={"config_list": []},
            human_input_mode="NEVER",
            **kwargs,
        )
        self.root_dir = root_dir

    def _safe_join(self, relative_path: str) -> str:
        rel = relative_path.strip().lstrip("/").replace("..", "")
        return os.path.join(self.root_dir, rel)

    def _apply_file_writes(self, text: str) -> List[str]:
        logs: List[str] = []
        for match in self.FILE_PATTERN.finditer(text):
            rel_path = match.group(1).strip()
            content = match.group(2)
            full_path = self._safe_join(rel_path)
            os.makedirs(os.path.dirname(full_path), exist_ok=True)
            with open(full_path, "w", encoding="utf-8") as f:
                f.write(content)
            logs.append(f"Écrit fichier: {rel_path} (len={len(content)} caractères)")
        return logs

    def _apply_file_reads(self, text: str) -> List[str]:
        logs: List[str] = []
        for match in self.READ_PATTERN.finditer(text):
            rel_path = match.group(1).strip()
            full_path = self._safe_join(rel_path)
            if not os.path.exists(full_path):
                logs.append(f"READ_FILE: {rel_path} introuvable.")
                continue
            with open(full_path, "r", encoding="utf-8") as f:
                content = f.read()
            logs.append(f"READ_FILE: {rel_path}\n--- CONTENU ---\n{content}\n--- FIN CONTENU ---")
        return logs

    def generate_reply(
        self,
        messages: List[Dict[str, Any]],
        sender: "ConversableAgent",
        **kwargs: Any,
    ) -> Dict[str, Any]:
        last = messages[-1]["content"]
        logs: List[str] = []

        writes = self._apply_file_writes(last)
        reads = self._apply_file_reads(last)

        if not writes and not reads:
            logs.append("Aucune instruction FILE:/READ_FILE: détectée dans le message.")
        else:
            logs.extend(writes)
            logs.extend(reads)

        return {
            "role": "assistant",
            "content": "\n\n".join(logs),
        }

# ------------------------------------------------------------------------------
# 1. Bridge AutoGen -> Qwen Code en tmux
#    (Custom "model client" simplifié compatible avec AutoGen 0.2.x)
# ------------------------------------------------------------------------------

class QwenTmuxModelClient:
    """
    Client minimal pour AutoGen qui envoie les messages au QwenTmuxLLM
    (donc à Qwen Code qui tourne dans tmux).

    Il respecte juste assez le "ModelClient protocol" pour marcher avec AutoGen:
    - create(params)
    - message_retrieval(response)
    - cost(response)
    - get_usage(response)
    """

    def __init__(self, config, **kwargs):
        # config_list[0] contient notre config custom
        cfg = None
        if isinstance(config, dict) and "config_list" in config:
            cfg = config["config_list"][0]
        else:
            cfg = config or {}

        self.session_name = cfg.get("session_name", "qwen_dev")
        self.wait_seconds = cfg.get("wait_seconds", 10)
        self.system_prompt = cfg.get("system_prompt", "")

        # On réutilise ton backend existant
        self.qwen = QwenTmuxLLM(
            session_name=self.session_name,
            system_prompt=self.system_prompt,
            wait_seconds=self.wait_seconds,
        )

    # Ce que AutoGen appelle pour "générer une réponse"
    def create(self, params):
        """
        params["messages"] = liste de dicts style OpenAI:
        [{"role": "user"|"assistant"|"system", "content": "..."}]
        """

        messages = params.get("messages", [])

        # On compacte simplement tout le contexte texte dans un prompt unique
        # pour Qwen Code (qui ne gère pas nativement les rôles).
        prompt_lines = []
        for m in messages:
            role = m.get("role", "user")
            content = m.get("content", "")
            prompt_lines.append(f"{role.upper()}:\n{content}\n")

        prompt = "\n".join(prompt_lines).strip()

        # Appel à ton Qwen CLI via tmux
        reply_text = self.qwen.chat(
            [
                {
                    "role": "user",
                    "content": prompt,
                }
            ]
        )

        # On emballe dans un objet façon OpenAI
        msg = SimpleNamespace()
        msg.content = reply_text
        msg.function_call = None

        choice = SimpleNamespace()
        choice.message = msg

        resp = SimpleNamespace()
        resp.choices = [choice]
        resp.model = "qwen-tmux"

        return resp

    def message_retrieval(self, response):
        # AutoGen utilise ça pour récupérer le texte
        return [c.message.content for c in response.choices]

    def cost(self, response) -> float:
        # Pas de coût pour Qwen CLI
        return 0.0

    @staticmethod
    def get_usage(response):
        # Pas de métriques d'usage pour l’instant
        return {}

class QwenTmuxModelClient:
    def __init__(self, config, **kwargs):
        cfg = None
        if isinstance(config, dict) and "config_list" in config:
            cfg = config["config_list"][0]
        else:
            cfg = config or {}

        self.session_name = cfg.get("session_name", "qwen_dev")
        self.wait_seconds = cfg.get("wait_seconds", 10)
        self.system_prompt = cfg.get("system_prompt", "")

        self.qwen = QwenTmuxLLM(
            session_name=self.session_name,
            system_prompt=self.system_prompt,
            wait_seconds=self.wait_seconds,
        )

    def create(self, params):
        messages = params.get("messages", [])

        prompt_lines = []
        for m in messages:
            role = m.get("role", "user")
            content = m.get("content", "")
            prompt_lines.append(f"{role.upper()}:\n{content}\n")

        prompt = "\n".join(prompt_lines).strip()

        reply_text = self.qwen.chat(
            [
                {
                    "role": "user",
                    "content": prompt,
                }
            ]
        )

        msg = SimpleNamespace()
        msg.content = reply_text
        msg.function_call = None

        choice = SimpleNamespace()
        choice.message = msg

        resp = SimpleNamespace()
        resp.choices = [choice]
        resp.model = "qwen-tmux"

        return resp

    def message_retrieval(self, response):
        return [c.message.content for c in response.choices]

    def cost(self, response) -> float:
        return 0.0

    @staticmethod
    def get_usage(response):
        return {}


def make_llm_config(session_name: str, system_prompt: str, wait_seconds: int = 10):

                "session_name": session_name,
                "system_prompt": system_prompt,
